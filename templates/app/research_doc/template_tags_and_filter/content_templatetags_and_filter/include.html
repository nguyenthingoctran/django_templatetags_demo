include
Load template và render nó ra

Render file html với URL string cụ thể

{% include "foo/bar.html" %}

Render file html với URL là một biến truyền vào

template_name = 'foo/bar.html'
{% include template_name %}

Render file html cùng với các đối số

{% include "name_snippet.html" with person="Jane" greeting="Hello" %}

Render file html cùng với đối số duy nhất

{% include "name_snippet.html" with greeting="Hi" only %}
