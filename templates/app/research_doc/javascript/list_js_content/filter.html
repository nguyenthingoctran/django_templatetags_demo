Filter
Trả về một mảng mới với các element đã được kiểm tra.

Code mẫu

<p id="show-data-filter"></p>

<script>
  const number_filter = [45, 4, 9, 16, 25];
  const over18_filter = number_filter.filter(function myFunction(value) {
    return value > 18
  });

  $("#show-data-filter").append(over18_filter + "<br/>")
</script>
