Định nghĩa: Phương thức này có thể thêm hoặc thay đổi 1 property, 1 property metadata, getters and setters của object.

Syntax

Object.defineProperty(object, property, descriptor)
1. Thêm một property

Code mẫu

<p id="show-data-dpa"></p>

<script>
  const dpa_person = {
    firstName: "John",
    lastName: "Doe",
    language: "EN"
  }

  Object.defineProperty(dpa_person, "year", { value: "2008" })
  $("#show-data-dpa").append(dpa_person.year)
</script>

3. Ẩn property

Code mẫu

<p id="show-data-hideP"></p>

<script>
  const hideP_person = {
      firstName: "John",
      lastName : "Doe",
      language : "NO",
  }

  Object.defineProperty(hideP_person, "language", {
    value: "EN",
    writable : true,
    enumerable : false,
    configurable : true
  });

  let hideP_txt = "";
  for (let x in hideP_person) {
    hideP_txt += hideP_person[x] + "
";
  }

  $("#show-data-hideP").append(hideP_txt)
</script>

4. Tạo setters và getters

Code mẫu

<p id="show-data-dpsg-person"></p>
<script>
  const dpsg_person = {
    firstName: "John",
    lastName : "Doe",
    language : "NO"
  };

  Object.defineProperty(dpsg_person, "language", {
    get : function() { return language },
    set : function(value) { language = value.toUpperCase()}
  });

  dpsg_person.language = "en";
  $("#show-data-dpsg-person").append(dpsg_person.language)
</script>
