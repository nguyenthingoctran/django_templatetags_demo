Định nghĩa: Hàm này trả về mô tả của một property.

Code mẫu

<p id="show-data-object-getOwnPropertyDescriptor"></p>

<script>
  const person_4 = {
    firstName: "John",
    lastName: "Doe",
    age: 50,
    eyeColor: "blue"
  };

  let descriptor = Object.getOwnPropertyDescriptor(person_4, "age");
  $("#show-data-object-getOwnPropertyDescriptor").append(descriptor.value + " " + descriptor.writable);
</script>
