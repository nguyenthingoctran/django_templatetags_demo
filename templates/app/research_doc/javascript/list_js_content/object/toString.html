3. Sử dụng Object.toString trong 1 object

Code mẫu

<p id="show-data-object-toString-2"></p>

<script>
  const person_13 = {
    firstName: "John",
    lastName: "Doe",
    age: 50,
    eyeColor: "blue"
  }

  const keys_2 = Object.toString(person_13)
  $("#show-data-object-toString-2").append(keys_2)
</script>
