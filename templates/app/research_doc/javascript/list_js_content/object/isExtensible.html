  const person_7 = { firstName: "John", lastName: "Doe" };

  var is_extensible = Object.isExtensible(person_7);
  $("#show-data-object-isExtensible").append("1) " + is_extensible + "
")

  Object.preventExtensions(person_7)
  is_extensible = Object.isExtensible(person_7);
  $("#show-data-object-isExtensible").append("2) " + is_extensible)
</script>
