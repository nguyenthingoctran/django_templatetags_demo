Định nghĩa: Trả về true nếu object là frozen.

Code mẫu

<p id="show-data-object-isFrozen"></p>

<script>
  const person_8 = { firstName: "John", lastName: "Doe" };
  Object.freeze(person_8)

  let text_3 = Object.isFrozen(person_8);
  $("#show-data-object-isFrozen").append(text_3)
</script>
