1. Trường hợp 2 object

Data mẫu

<script>
  const a_target = { a : 1, b : 2 };

  const a_returnedTarget = Object.assign(a_target, a_source)
  $("#show-data-assign").append(JSON.stringify(a_returnedTarget))
</script>

5. Merging object

Data mẫu

<p id="show-data-assign-mo"></p>
<script>
  const o1_mo = { a : 1 };
  const o2_mo = { b : 2 };
  const o3_mo = { c : 3 };

  const obj_mo = Object.assign(o1_mo, o2_mo, o3_mo)

6. Merging object with the properties

Code mẫu

<p id="show-data-assign-mowsp"></p>
<script>
  const o1_mowsp = { a : 1, b : 1, c : 1 };
  const o2_mowsp = { b : 2, c : 2 };
  const o3_mowsp = { c : 3 };

  const obj_mowsp = Object.assign({}, o1_mowsp, o2_mowsp, o3_mowsp)
  $("#show-data-assign-mowsp").append(JSON.stringify(obj_mowsp))
</script>
