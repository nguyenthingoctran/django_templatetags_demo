Reduce
Xử lý mảng thành 1 giá trị theo yêu cầu.

<script>
  const numbers_reduce = [45, 4, 9, 16, 25];

  let sum = numbers_reduce.reduce(function myFunction(total, value) {
    return total + value
  });

  $("#show-data-reduce").append(sum)
</script>
