Object Create
Tạo một Object khác từ 1 Object có sẵn

Code mẫu

<h1 id="show-data-object-create"></h1>

<script>
  const person_oc = {
    firstName: "John",
    lastName: "Doe"
  }

  const man = Object.create(person_oc);
  man.firstName = 'Peter';

  $("#show-data-object-create").append(person_oc.firstName + " and " + man.firstName)
</script>
