Every
Trả về giá trị true/false khi duyệt qua all element trong mảng.

Code mẫu

<h1 id="show-data-every"></h1>

<script>
  const numbers_every = [45, 4, 9, 16, 25];
  let allOver18 = numbers_every.every(function myFunction(value) {
    return value > 18
  });

  if (allOver18) {
    allOver18 = "true"
  } else {
    allOver18 = "false"
  }

  $("#show-data-every").append(allOver18)
</script>
